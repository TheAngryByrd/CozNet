cmake_minimum_required(VERSION 3.20)
project(CozNetService LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(CozNet.ServiceCore STATIC ServiceOwner.cpp TargetLease.cpp)
set_target_properties(CozNet.ServiceCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(CozNet.ServiceCore PUBLIC cxx_std_20)
target_link_libraries(CozNet.ServiceCore PUBLIC Threads::Threads)
add_executable(ServiceOwnerChecks OwnerChecks.cpp)
add_executable(TargetLeaseChecks TargetLeaseChecks.cpp)
add_executable(LeaseMailboxChecks LeaseMailboxChecks.cpp)
add_executable(TimerJournalSnapshotChecks TimerJournalSnapshotChecks.cpp)
target_link_libraries(TimerJournalSnapshotChecks PRIVATE CozNet.ServiceCore)
target_link_libraries(LeaseMailboxChecks PRIVATE CozNet.ServiceCore)
target_link_libraries(TargetLeaseChecks PRIVATE CozNet.ServiceCore)
target_link_libraries(ServiceOwnerChecks PRIVATE CozNet.ServiceCore)
if(MSVC)
  target_compile_options(CozNet.ServiceCore PRIVATE /W4 /WX)
  target_compile_options(ServiceOwnerChecks PRIVATE /W4 /WX)
  target_compile_options(TargetLeaseChecks PRIVATE /W4 /WX)
  target_compile_options(LeaseMailboxChecks PRIVATE /W4 /WX)
  target_compile_options(TimerJournalSnapshotChecks PRIVATE /W4 /WX)
else()
  target_compile_options(CozNet.ServiceCore PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ServiceOwnerChecks PRIVATE -Wall -Wextra -Werror)
  target_compile_options(TargetLeaseChecks PRIVATE -Wall -Wextra -Werror)
  target_compile_options(LeaseMailboxChecks PRIVATE -Wall -Wextra -Werror)
  target_compile_options(TimerJournalSnapshotChecks PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME timer-journal-snapshot COMMAND TimerJournalSnapshotChecks)
set_tests_properties(timer-journal-snapshot PROPERTIES TIMEOUT 30)
add_test(NAME lease-mailbox COMMAND LeaseMailboxChecks)
set_tests_properties(lease-mailbox PROPERTIES TIMEOUT 30)
add_test(NAME target-lease COMMAND TargetLeaseChecks)
set_tests_properties(target-lease PROPERTIES TIMEOUT 30)
add_test(NAME service-owner COMMAND ServiceOwnerChecks)
set_tests_properties(service-owner PROPERTIES TIMEOUT 30)
foreach(mode ambiguous-reused ambiguous-zero sent-reused rejected-limit)
  add_test(NAME service-owner-${mode} COMMAND ServiceOwnerChecks ${mode})
  set_tests_properties(service-owner-${mode} PROPERTIES TIMEOUT 30)
endforeach()

  target_compile_options(TimerDeliveryChecks PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(TimerDeliveryChecks PRIVATE CozNet.ServiceCore Threads::Threads)
  add_executable(TimerReuseChecks TimerReuseChecks.Linux.cpp TimerDeliveryWorkload.LinuxX64.S)

  target_link_libraries(CozNet.ServiceSignal PUBLIC CozNet.ServiceCore CozNet.Signal)

  target_compile_options(SignalServiceChecks PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(SignalServiceChecks PRIVATE CozNet.ServiceSignal)
  foreach(mode lease-spin lease-sleep selected pending-cancel restore-live occupied range)
    add_test(NAME signal-service-${mode} COMMAND SignalServiceChecks ${mode} ${CMAKE_CURRENT_BINARY_DIR}/${mode}.tsv)
    set_tests_properties(signal-service-${mode} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
