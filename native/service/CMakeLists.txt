cmake_minimum_required(VERSION 3.20)
project(CozNetService LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(CozNet.ServiceCore STATIC ServiceOwner.cpp)
set_target_properties(CozNet.ServiceCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(CozNet.ServiceCore PUBLIC cxx_std_20)
target_link_libraries(CozNet.ServiceCore PUBLIC Threads::Threads)
add_executable(ServiceOwnerChecks OwnerChecks.cpp)
target_link_libraries(ServiceOwnerChecks PRIVATE CozNet.ServiceCore)
if(MSVC)
  target_compile_options(CozNet.ServiceCore PRIVATE /W4 /WX)
  target_compile_options(ServiceOwnerChecks PRIVATE /W4 /WX)
else()
  target_compile_options(CozNet.ServiceCore PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ServiceOwnerChecks PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME service-owner COMMAND ServiceOwnerChecks)
set_tests_properties(service-owner PROPERTIES TIMEOUT 30)
foreach(mode ambiguous-reused ambiguous-zero sent-reused rejected-limit)
  add_test(NAME service-owner-${mode} COMMAND ServiceOwnerChecks ${mode})
  set_tests_properties(service-owner-${mode} PROPERTIES TIMEOUT 30)
endforeach()

option(COZ_SERVICE_LINUX_SIGNAL "Build the Linux x64 signal transport checks" OFF)
if(COZ_SERVICE_LINUX_SIGNAL)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    message(FATAL_ERROR "The service signal transport requires Linux x64.")
  endif()
  enable_language(ASM)
  add_subdirectory(../signal signal)
  add_library(CozNet.ServiceSignal STATIC SignalTransport.Linux.cpp)
  set_target_properties(CozNet.ServiceSignal PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_features(CozNet.ServiceSignal PUBLIC cxx_std_20)
  target_compile_options(CozNet.ServiceSignal PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(CozNet.ServiceSignal PUBLIC CozNet.ServiceCore CozNet.Signal)
  add_library(CozNet.Service SHARED ServiceApi.Linux.cpp)
  target_compile_definitions(CozNet.Service PRIVATE COZ_SERVICE_BUILD)
  target_compile_options(CozNet.Service PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
  target_link_libraries(CozNet.Service PRIVATE CozNet.ServiceSignal)
  add_executable(ServiceApiChecks ApiChecks.cpp)
  target_compile_features(ServiceApiChecks PRIVATE cxx_std_20)
  target_compile_options(ServiceApiChecks PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(ServiceApiChecks PRIVATE CozNet.Service Threads::Threads)
  add_test(NAME service-abi COMMAND ServiceApiChecks)
  set_tests_properties(service-abi PROPERTIES TIMEOUT 30)
  add_executable(SignalServiceChecks SignalServiceChecks.cpp SignalWorkload.LinuxX64.S)
  target_compile_options(SignalServiceChecks PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(SignalServiceChecks PRIVATE CozNet.ServiceSignal)
  foreach(mode lease-spin lease-sleep selected pending-cancel restore-live occupied range)
    add_test(NAME signal-service-${mode} COMMAND SignalServiceChecks ${mode} ${CMAKE_CURRENT_BINARY_DIR}/${mode}.tsv)
    set_tests_properties(signal-service-${mode} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
