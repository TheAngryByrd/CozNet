cmake_minimum_required(VERSION 3.20)
project(CozNetService LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(CozNet.ServiceCore STATIC ServiceOwner.cpp)
target_compile_features(CozNet.ServiceCore PUBLIC cxx_std_20)
target_link_libraries(CozNet.ServiceCore PUBLIC Threads::Threads)
add_executable(ServiceOwnerChecks OwnerChecks.cpp)
target_link_libraries(ServiceOwnerChecks PRIVATE CozNet.ServiceCore)
if(MSVC)
  target_compile_options(CozNet.ServiceCore PRIVATE /W4 /WX)
  target_compile_options(ServiceOwnerChecks PRIVATE /W4 /WX)
else()
  target_compile_options(CozNet.ServiceCore PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ServiceOwnerChecks PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME service-owner COMMAND ServiceOwnerChecks)
set_tests_properties(service-owner PROPERTIES TIMEOUT 30)
foreach(mode ambiguous-reused ambiguous-zero sent-reused rejected-limit)
  add_test(NAME service-owner-${mode} COMMAND ServiceOwnerChecks ${mode})
  set_tests_properties(service-owner-${mode} PROPERTIES TIMEOUT 30)
endforeach()
