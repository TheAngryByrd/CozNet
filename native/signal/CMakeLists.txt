cmake_minimum_required(VERSION 3.20)
project(CozNetSignal LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  message(FATAL_ERROR "The signal experiment requires Linux x64.")
endif()
add_library(CozNet.Signal SHARED Signal.LinuxX64.cpp)
target_compile_features(CozNet.Signal PRIVATE cxx_std_20)
target_compile_options(CozNet.Signal PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti -fvisibility=hidden)
find_package(Threads REQUIRED)
add_executable(SignalContractChecks ContractChecks.cpp)
target_compile_features(SignalContractChecks PRIVATE cxx_std_20)
target_compile_options(SignalContractChecks PRIVATE -Wall -Wextra -Werror)
target_link_libraries(SignalContractChecks PRIVATE CozNet.Signal Threads::Threads)
add_library(CozNet.SignalWorkload SHARED Workload.LinuxX64.cpp)
target_compile_features(CozNet.SignalWorkload PRIVATE cxx_std_20)
target_compile_options(CozNet.SignalWorkload PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
